#pragma once
#include <iostream>
#include <algorithm>
#include <stdexcept>
#include "String.h"

using llu = unsigned long long;

template <typename T>
class Vector {
private:
    T* array;
    int capacity;
    int size;

    void reallocate(int newCap) {
        if (newCap < 1) newCap = 1; // 🔥 đảm bảo không bao giờ = 0
        T* newArr = new T[newCap];
        for (int i = 0; i < size; ++i)
            newArr[i] = array[i];
        delete[] array;
        array = newArr;
        capacity = newCap;
    }

public:
    // ===== Constructors =====
    Vector() : array(new T[1]), capacity(1), size(0) {}

    Vector(int n, const T& val = T()) {
        capacity = std::max(1, n);
        size = n;
        array = new T[capacity];
        for (int i = 0; i < size; ++i)
            array[i] = val;
    }

    Vector(const Vector& other) {
        capacity = other.capacity;
        size = other.size;
        array = (capacity > 0) ? new T[capacity] : nullptr;
        for (int i = 0; i < size; ++i)
            array[i] = other.array[i];
    }

    Vector(Vector&& other) noexcept
        : array(other.array), capacity(other.capacity), size(other.size) {
        other.array = nullptr;
        other.capacity = 0;
        other.size = 0;
    }

    Vector& operator=(const Vector& other) {
        if (this == &other) return *this;
        delete[] array;
        capacity = other.capacity;
        size = other.size;
        array = (capacity > 0) ? new T[capacity] : nullptr;
        for (int i = 0; i < size; ++i)
            array[i] = other.array[i];
        return *this;
    }

    Vector& operator=(Vector&& other) noexcept {
        if (this != &other) {
            delete[] array;
            array = other.array;
            capacity = other.capacity;
            size = other.size;
            other.array = nullptr;
            other.capacity = 0;
            other.size = 0;
        }
        return *this;
    }

    // ===== Destructor =====
    ~Vector() {
        delete[] array;
        array = nullptr;
        capacity = 0;
        size = 0;
    }

    // ===== Core methods =====
    void resize(int newSize) {
        if (newSize < 0)
            throw std::out_of_range("Negative size not allowed");

        if (newSize <= size) {
            size = newSize;
            return;
        }

        if (newSize > capacity)
            reallocate(std::max(capacity * 2, newSize));

        for (int i = size; i < newSize; ++i)
            array[i] = T();

        size = newSize;
    }

    void resize(int newSize, const T& value) {
        if (newSize < 0)
            throw std::out_of_range("Negative size not allowed");

        if (newSize <= size) {
            size = newSize;
            return;
        }

        if (newSize > capacity)
            reallocate(std::max(capacity * 2, newSize));

        for (int i = size; i < newSize; ++i)
            array[i] = value;

        size = newSize;
    }

    void push_back(const T& value) {
        if (capacity == 0) reallocate(1);     // 🔥 đảm bảo không 0
        if (size >= capacity) reallocate(std::max(1, capacity * 2));
        array[size++] = value;
    }

    void pop_back() {
        if (size > 0) --size;
    }

    void erase(int index) {
        if (index < 0 || index >= size) return;
        for (int i = index; i < size - 1; ++i)
            array[i] = array[i + 1];
        --size;
    }

    T* erase(T* pos) {
        if (pos < array || pos >= array + size)
            return end();
        int index = static_cast<int>(pos - array);
        erase(index);
        return array + index;
    }

    T* insert(T* pos, const T& val) {
        if (pos < array || pos > array + size)
            return end();
        int index = static_cast<int>(pos - array);
        if (size >= capacity) reallocate(capacity * 2);
        for (int i = size; i > index; --i)
            array[i] = array[i - 1];
        array[index] = val;
        ++size;
        return array + index;
    }

    void clear() {
        size = 0; // không cần gọi destructor từng phần tử
    }

    void reverse() {
        for (int i = 0; i < size / 2; ++i)
            std::swap(array[i], array[size - i - 1]);
    }

    bool remove(const T& value) {
        for (int i = 0; i < size; ++i) {
            if (array[i] == value) {
                erase(i);
                return true;
            }
        }
        return false;
    }

    // ===== Info =====
    int get_size() const { return size; }
    int get_capacity() const { return capacity; }
    bool empty() const { return size == 0; }

    // ===== Operators =====
    T& operator[](int index) {
        if (index < 0 || index >= size)
            throw std::out_of_range("Vector index out of range");
        return array[index];
    }

    const T& operator[](int index) const {
        if (index < 0 || index >= size)
            throw std::out_of_range("Vector index out of range");
        return array[index];
    }

    bool operator==(const Vector& other) const {
        if (size != other.size) return false;
        for (int i = 0; i < size; ++i)
            if (!(array[i] == other.array[i])) return false;
        return true;
    }

    // ===== Iterators =====
    T* begin() { return array; }
    T* end() { return array + size; }
    const T* begin() const { return array; }
    const T* end() const { return array + size; }

    // ===== Front/Back =====
    T& front() {
        if (size == 0) throw std::out_of_range("Vector is empty");
        return array[0];
    }

    T& back() {
        if (size == 0) throw std::out_of_range("Vector is empty");
        return array[size - 1];
    }

    const T& front() const {
        if (size == 0) throw std::out_of_range("Vector is empty");
        return array[0];
    }

    const T& back() const {
        if (size == 0) throw std::out_of_range("Vector is empty");
        return array[size - 1];
    }
};

// ===== Explicit template instantiation =====
template class Vector<int>;
template class Vector<String>;
template class Vector<Vector<int>>;
